% 数学命令
\renewcommand{\va}[1]{\overrightarrow{#1}}
\renewcommand{\vb}[1]{\symbfit{#1}}
\newcommand{\ee}{\mathrm{e}} % 正体自然常数e
\newcommand{\ii}{\mathrm{i}} % 正体虚数i
\DeclarePairedDelimiter\floor{\lfloor}{\rfloor}
\let\grad\relax\DeclareMathOperator{\grad}{\symbf{grad}} % 散度
\DeclareMathOperator{\rot}{\symbf{rot}} % 旋度

% 圆弧符号
\DeclareSymbolFont{ugmL}{OMX}{mdugm}{m}{n}
\DeclareMathAccent{\wideparen}{\mathord}{ugmL}{"F3}

% 文本装饰命令
\newcommand{\overtext}[2]{
    \stackon[0.5pt]
    {\textbf{#1}}
    {\textsf{\textbf{\fontsize{5pt}{5pt}\selectfont #2}}}
}

\newcommand{\circnum}[1]{
    {\bfseries\itshape{#1}\,${}^\circ$}
}

\newcommand{\skipitem}[1][1]{\addtocounter{enumi}{#1}} % 跳过enumerate编号

% 图形符号命令
\newcommand{\dashedmid}{
    \mathrel{
        \tikz[baseline=0pt]{
            \draw[
                dashed,
                dash pattern=on 2.4pt off 2.4pt,
                line width=0.6pt
            ]
            (0,-0.55ex)--(0,1.85ex);
        }
    }
}

% 超链接图标命令
\newcommand{\wiki}[1]{
    \!\!\href{#1}{
        \raisebox{1.2ex}
        {\textcolor{black}{\scriptsize\faWikipediaW}}
    }\!\!\!\!
}

\newcommand{\video}[1]{
    \!\!\href{#1}{
        \raisebox{1.2ex}
        {\textcolor{black}{\scriptsize\faPlayCircle[regular]}}
    }\!\!\!\!
}
